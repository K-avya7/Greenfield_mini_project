-- ============================================================
-- ETL Pipeline: OLTP -> OLAP Star Schema
-- File: etl_oltp_to_olap.sql
--
-- Stored Procedures (run in order via sp_run_full_etl_olap):
--   SP-1 : sp_load_dim_date
--   SP-2 : sp_load_dim_department
--   SP-3 : sp_load_dim_project
--   SP-4 : sp_load_dim_employee       <- SCD Type 2 core
--   SP-5 : sp_load_fact_reviews       <- SCD2 date-range join
--   SP-6 : sp_run_full_etl_olap       <- Master orchestrator
-- ============================================================

USE hr_analytics_dw;

-- ============================================================
-- SP-1: sp_load_dim_date
-- Generates one row per calendar day from 2010-01-01 to 2030-12-31
-- No CTE needed: pure iterative date generation (7,305 rows)
-- ============================================================
DROP PROCEDURE IF EXISTS sp_load_dim_date;

DELIMITER $$
CREATE PROCEDURE sp_load_dim_date()
BEGIN
    DECLARE v_date DATE DEFAULT '2010-01-01';
    DECLARE v_end  DATE DEFAULT '2030-12-31';

    TRUNCATE TABLE dim_date;

    WHILE v_date <= v_end DO
        INSERT INTO dim_date (
            date_sk, full_date, day, month,
            month_name, quarter, year, is_weekend
        ) VALUES (
            CAST(DATE_FORMAT(v_date, '%Y%m%d') AS UNSIGNED), -- YYYYMMDD key
            v_date,
            DAY(v_date),
            MONTH(v_date),
            MONTHNAME(v_date),
            QUARTER(v_date),
            YEAR(v_date),
            IF(DAYOFWEEK(v_date) IN (1,7), 1, 0) -- 1=Sun, 7=Sat
        );
        SET v_date = DATE_ADD(v_date, INTERVAL 1 DAY);
    END WHILE;

    SELECT CONCAT('SP-1 dim_date: ', COUNT(*), ' rows loaded') AS etl_log
    FROM dim_date;
END$$
DELIMITER ;


-- ============================================================
-- SP-2: sp_load_dim_department
-- Source  : OLTP departments table
-- CTE     : ranked_depts  - deduplicates and trims names
-- Window  : ROW_NUMBER() PARTITION BY department_name
-- ============================================================
DROP PROCEDURE IF EXISTS sp_load_dim_department;

DELIMITER $$
CREATE PROCEDURE sp_load_dim_department()
BEGIN
    TRUNCATE TABLE dim_department;

    INSERT INTO dim_department (department_id, department_name)
    WITH ranked_depts AS (
        SELECT
            department_id,
            TRIM(department_name)              AS department_name,
            -- Window Fn: deduplicate any accidental duplicates
            ROW_NUMBER() OVER (
                PARTITION BY TRIM(department_name)
                ORDER BY department_id ASC
            )                                  AS rn
        FROM departments
        WHERE department_name IS NOT NULL
    )
    SELECT department_id, department_name
    FROM   ranked_depts
    WHERE  rn = 1
    ORDER  BY department_id;

    SELECT CONCAT('SP-2 dim_department: ', COUNT(*), ' rows loaded') AS etl_log
    FROM dim_department;
END$$
DELIMITER ;

-- ============================================================
-- SP-3: sp_load_dim_project
-- Source  : OLTP projects table
-- If no projects exist, inserts one default "No Project" record
-- so that fact rows with project_sk = NULL still satisfy the FK
-- ============================================================
DROP PROCEDURE IF EXISTS sp_load_dim_project;

DELIMITER $$
CREATE PROCEDURE sp_load_dim_project()
BEGIN
    DECLARE v_count INT DEFAULT 0;
    SELECT COUNT(*) INTO v_count FROM projects;

    TRUNCATE TABLE dim_project;

    IF v_count > 0 THEN
        INSERT INTO dim_project (project_id, project_name, start_date, end_date)
        SELECT
            project_id,
            TRIM(project_name),
            start_date,
            end_date
        FROM projects
        ORDER BY project_id;
    ELSE
        -- Default sentinel: project_sk = 1, project_id = -1
        INSERT INTO dim_project (project_id, project_name, start_date, end_date)
        VALUES (-1, 'No Project Assigned', NULL, NULL);
    END IF;

    SELECT CONCAT('SP-3 dim_project: ', COUNT(*), ' rows loaded') AS etl_log
    FROM dim_project;
END$$
DELIMITER ;


-- ============================================================
-- SP-4: sp_load_dim_employee   *** SCD Type 2 Core ***
--
-- Sources:
--   employee_job_history  -> SCD2 version data (employee_sk,
--                            job_level, income, rates, timeline)
--   employees             -> Fixed personal attributes
--                            (name, gender, education_field)
--   job_roles             -> job_role_name (text) for the dimension
--   staging_employees     -> manager_id per version (historical)
--   dim_department        -> Resolves department_sk (outrigger FK)
--
-- CTEs:
--   all_versions     - joins all sources into one flat row per version
--   ranked_versions  - ROW_NUMBER() oldest->newest per employee
--   change_detection - LAG() to detect what changed vs prior version
--   dept_resolved    - resolves department_sk from dim_department
--
-- Window Functions:
--   ROW_NUMBER() OVER (PARTITION BY employee_id ORDER BY start_date)
--   COUNT()      OVER (PARTITION BY employee_id)
--   LAG()        OVER x3 (department, job_role, monthly_income)
-- ============================================================
DROP PROCEDURE IF EXISTS sp_load_dim_employee;

DELIMITER $$
CREATE PROCEDURE sp_load_dim_employee()
BEGIN
    TRUNCATE TABLE dim_employee;

    INSERT INTO dim_employee (
        employee_sk,
        employee_id,
        department_sk,
        first_name,
        last_name,
        gender,
        education_field,
        job_role,
        job_level,
        monthly_income,
        daily_rate,
        hourly_rate,
        manager_id,
        change_reason,
        effective_start_date,
        effective_end_date,
        is_current
    )
    -- ── CTE 1 : Assemble all version data from OLTP tables ──────────
    WITH all_versions AS (
        SELECT
            h.employee_sk,
            h.employee_id,
            -- Fixed attributes from employees table
            e.first_name,
            e.last_name,
            e.gender,
            e.education_field,
            -- Department reference (will be resolved to SK in CTE 4)
            h.department_id,
            -- Job role name from job_roles lookup
            j.job_role_name                         AS job_role,
            h.job_level,
            -- SCD2-tracked compensation attributes
            h.monthly_income,
            h.daily_rate,
            h.hourly_rate,
            -- manager_id: join staging via employee_sk for historical accuracy
            s.manager_id,
            -- SCD2 audit and control
            COALESCE(TRIM(h.change_reason),
                     'Initial Onboarding')          AS change_reason,
            h.effective_start_date,
            h.effective_end_date,
            h.is_current
        FROM employee_job_history h
        -- Get fixed personal attributes (current employee record)
        JOIN employees   e  ON  e.employee_id  = h.employee_id
        -- Get job role text from lookup
        JOIN job_roles   j  ON  j.job_role_id  = h.job_role_id
        -- Get historical manager_id per SCD2 version from staging
        JOIN staging_employees s ON s.employee_sk = h.employee_sk
        WHERE h.effective_start_date IS NOT NULL
    ),

    -- ── CTE 2 : Rank versions oldest -> newest per employee ──────────
    ranked_versions AS (
        SELECT
            av.*,
            -- Window Fn 1: sequence number for this employee's versions
            ROW_NUMBER() OVER (
                PARTITION BY av.employee_id
                ORDER BY av.effective_start_date ASC
            )                                       AS version_num,
            -- Window Fn 2: how many total versions does this employee have?
            COUNT(*) OVER (
                PARTITION BY av.employee_id
            )                                       AS total_versions
        FROM all_versions av
    ),

    -- ── CTE 3 : Detect what changed vs the previous version (LAG) ───
    change_detection AS (
        SELECT
            rv.*,
            -- Window Fn 3: what was the department in the prior version?
            LAG(rv.department_id) OVER (
                PARTITION BY rv.employee_id
                ORDER BY rv.effective_start_date
            )                                       AS prev_department_id,
            -- Window Fn 4: what was the job role in the prior version?
            LAG(rv.job_role) OVER (
                PARTITION BY rv.employee_id
                ORDER BY rv.effective_start_date
            )                                       AS prev_job_role,
            -- Window Fn 5: what was the income in the prior version?
            LAG(rv.monthly_income) OVER (
                PARTITION BY rv.employee_id
                ORDER BY rv.effective_start_date
            )                                       AS prev_income
        FROM ranked_versions rv
    ),

    -- ── CTE 4 : Resolve department_sk (Outrigger FK) ─────────────────
    dept_resolved AS (
        SELECT
            cd.*,
            -- Join dim_department to get the OLAP surrogate key
            dd.department_sk
        FROM change_detection cd
        JOIN dim_department dd ON dd.department_id = cd.department_id
    )

    -- ── Final projection into dim_employee ───────────────────────────
    SELECT
        dr.employee_sk,
        dr.employee_id,
        dr.department_sk,          -- outrigger FK resolved via CTE 4
        dr.first_name,
        dr.last_name,
        dr.gender,
        dr.education_field,
        dr.job_role,
        dr.job_level,
        dr.monthly_income,
        dr.daily_rate,
        dr.hourly_rate,
        dr.manager_id,
        dr.change_reason,
        dr.effective_start_date,
        dr.effective_end_date,
        dr.is_current
    FROM dept_resolved dr
    ORDER BY dr.employee_id, dr.effective_start_date;

    SELECT CONCAT('SP-4 dim_employee: ', COUNT(*), ' SCD2 version rows loaded') AS etl_log
    FROM dim_employee;
END$$
DELIMITER ;


-- ============================================================
-- SP-5: sp_load_fact_reviews
--
-- Source  : OLTP reviews table (employee_id, review_date,
--                                performance_rating)
--
-- CTEs:
--   base_reviews     - clean reviews, cast date_sk, guard invalid dates
--   ranked_reviews   - DENSE_RANK() sequences reviews per employee
--   dept_performance - AVG() and RANK() OVER for analytics enrichment
--   scd2_resolved    - THE critical SCD2 date-range join to get the
--                      exact employee_sk version active on review_date
--
-- Window Functions:
--   DENSE_RANK() OVER (PARTITION BY employee_id ORDER BY review_date)
--   AVG()        OVER (PARTITION BY department_sk)
--   RANK()       OVER (PARTITION BY department_sk ORDER BY rating DESC)
--
-- SCD2 Key Rule:
--   JOIN dim_employee WHERE review_date BETWEEN effective_start_date
--                                           AND effective_end_date
--   This picks the employee's VERSION that was active at review time.
-- ============================================================
DROP PROCEDURE IF EXISTS sp_load_fact_reviews;

DELIMITER $$
CREATE PROCEDURE sp_load_fact_reviews()
BEGIN
    TRUNCATE TABLE fact_performance_reviews;

    INSERT INTO fact_performance_reviews (
        employee_sk,
        department_sk,
        project_sk,
        date_sk,
        review_id,
        performance_rating
    )
    -- ── CTE 1 : Clean reviews + cast date to integer date_sk ─────────
    WITH base_reviews AS (
        SELECT
            r.review_id,
            r.employee_id,
            r.review_date,
            r.performance_rating,
            -- Cast review_date to YYYYMMDD integer to match dim_date.date_sk
            CAST(DATE_FORMAT(r.review_date, '%Y%m%d') AS UNSIGNED)  AS date_sk
        FROM reviews r
        WHERE r.review_date        IS NOT NULL
          AND r.performance_rating IS NOT NULL
          -- Guard: only dates that exist in dim_date are valid
          AND EXISTS (
                SELECT 1
                FROM dim_date dd
                WHERE dd.full_date = r.review_date
          )
    ),

    -- ── CTE 2 : Rank each employee's reviews chronologically ─────────
    ranked_reviews AS (
        SELECT
            br.*,
            -- Window Fn 1: review number per employee (1 = first review)
            DENSE_RANK() OVER (
                PARTITION BY br.employee_id
                ORDER BY br.review_date ASC
            )                                       AS review_sequence
        FROM base_reviews br
    ),

    -- ── CTE 3 : SCD2 Date-Range Join — the KEY step ──────────────────
    -- Matches each review to the EXACT employee version active on that date
    scd2_resolved AS (
        SELECT
            de.employee_sk,       -- CORRECT historical version of the employee
            de.department_sk,     -- department the employee was in at review time
            rr.date_sk,
            rr.review_id,
            rr.performance_rating,
            rr.review_sequence
        FROM ranked_reviews rr
        -- SCD2 Rule: find version where review_date falls within the version period
        JOIN dim_employee de
            ON  de.employee_id        = rr.employee_id
            AND rr.review_date BETWEEN de.effective_start_date
                                   AND de.effective_end_date
    ),

    -- ── CTE 4 : Add analytical window measures (for enrichment) ──────
    enriched AS (
        SELECT
            sr.*,
            -- Window Fn 2: average rating across the whole department
            AVG(sr.performance_rating) OVER (
                PARTITION BY sr.department_sk
            )                                       AS dept_avg_rating,
            -- Window Fn 3: rank employee within their department by rating
            RANK() OVER (
                PARTITION BY sr.department_sk
                ORDER BY sr.performance_rating DESC
            )                                       AS dept_rating_rank
        FROM scd2_resolved sr
    )

    -- ── Final projection into fact table ─────────────────────────────
    SELECT
        e.employee_sk,
        e.department_sk,
        NULL            AS project_sk,   -- no project data in staging/OLTP
        e.date_sk,
        e.review_id,
        e.performance_rating
    FROM enriched e
    ORDER BY e.employee_sk, e.date_sk;

    SELECT CONCAT('SP-5 fact_performance_reviews: ', COUNT(*), ' rows loaded') AS etl_log
    FROM fact_performance_reviews;
END$$
DELIMITER ;


-- ============================================================
-- SP-6: sp_run_full_etl_olap  (Master Orchestrator)
-- Calls all 5 SPs in the correct dependency order:
--   dim_date -> dim_department -> dim_project
--   -> dim_employee (needs dim_department)
--   -> fact_performance_reviews (needs all dims)
-- ============================================================
DROP PROCEDURE IF EXISTS sp_run_full_etl_olap;

DELIMITER $$
CREATE PROCEDURE sp_run_full_etl_olap()
BEGIN
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        GET DIAGNOSTICS CONDITION 1
            @err_msg  = MESSAGE_TEXT,
            @err_code = MYSQL_ERRNO;
        SELECT CONCAT('[FAILED] Error ', @err_code, ': ', @err_msg) AS etl_error;
    END;

    SELECT '======================================' AS '';
    SELECT '  HR Analytics DW - OLAP ETL Start   ' AS '';
    SELECT '======================================' AS '';

    SELECT 'Step 1/5 | Loading dim_date ...'        AS etl_step; CALL sp_load_dim_date();
    SELECT 'Step 2/5 | Loading dim_department ...'  AS etl_step; CALL sp_load_dim_department();
    SELECT 'Step 3/5 | Loading dim_project ...'     AS etl_step; CALL sp_load_dim_project();
    SELECT 'Step 4/5 | Loading dim_employee ...'    AS etl_step; CALL sp_load_dim_employee();
    SELECT 'Step 5/5 | Loading fact_reviews ...'    AS etl_step; CALL sp_load_fact_reviews();

    SELECT '======================================' AS '';
    SELECT '  OLAP ETL Complete - Final Counts    ' AS '';
    SELECT '======================================' AS '';

    -- Final verification counts
    SELECT 'dim_date'                  AS olap_table, COUNT(*) AS row_count FROM dim_date
    UNION ALL SELECT 'dim_department',               COUNT(*)              FROM dim_department
    UNION ALL SELECT 'dim_project',                  COUNT(*)              FROM dim_project
    UNION ALL SELECT 'dim_employee (SCD2)',           COUNT(*)              FROM dim_employee
    UNION ALL SELECT 'fact_performance_reviews',     COUNT(*)              FROM fact_performance_reviews;
END$$
DELIMITER ;
